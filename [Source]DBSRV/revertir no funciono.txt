	STRUCT_GUILD guild;

	int ret = ReadGuildFile(nID, &guild);	
	if(ret==FALSE) return FALSE;

	guild.dwTime[0] = dwTime1;
	guild.dwTime[1] = dwTime2;
	guild.dwTime[2] = dwTime3;

	ret = WriteGuildFile(nID, &guild);
	if (ret==FALSE) return FALSE;

	return TRUE;
}

	return TRUE;
